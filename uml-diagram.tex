\section{State Chart Diagram}
Figure~\ref{fig:statechart} shows the lifecycle of the AI Agent for a single user command, from idle to completion or error recovery.

\begin{figure}[H]
\centering
\resizebox{0.88\textwidth}{!}{%
\begin{tikzpicture}[
    state/.style={draw, rounded rectangle, minimum width=3.8cm,
                  minimum height=1.0cm, align=center,
                  fill=blue!10, font=\small},
    init/.style={draw, circle, fill=black, minimum size=9pt, inner sep=0},
    final/.style={draw, circle, fill=black, minimum size=9pt, inner sep=0,
                  double, double distance=2.5pt},
    trans/.style={-Stealth, thick},
    lbl/.style={font=\scriptsize, fill=white, inner sep=2pt}
]

% ---- States ----
\node[init]  (S)    at (0,    0)     {};
\node[state] (IDLE) at (0,   -2.5)  {Idle};
\node[state] (RECV) at (0,   -5.5)  {Receiving Input};
\node[state] (STT)  at (-7.5,-9.0)  {STT Processing\\(voice $\rightarrow$ text)};
\node[state] (PLAN) at (0,   -9.0)  {LLM Planning};
\node[state] (ERR)  at (7.5, -9.0)  {Error State};
\node[state] (EXEC) at (0,  -12.5)  {Executing Actions\\via CDP};
\node[state] (WAIT) at (7.5,-12.5)  {Waiting for\\Page Load};
\node[state] (DONE) at (0,  -16.0)  {Completed};
\node[final] (F)    at (0,  -18.0)  {};

% ---- Transitions ----

% Start → Idle
\draw[trans] (S) -- (IDLE);

% Idle → Receiving Input
\draw[trans] (IDLE) -- node[lbl, right]{user submits command} (RECV);

% Receiving Input → LLM Planning  (text, straight down offset slightly right)
\draw[trans] (RECV.south) -- node[lbl, right]{text command} (PLAN.north);

% Receiving Input → STT Processing  (voice, bypass left of STT box → enter from west)
\draw[trans] (RECV.west) --
    node[lbl, above, pos=0.5]{voice detected}
    ++(-8.6, 0) |- (STT.west);

% STT → LLM Planning  (horizontal; label centred below arrow)
\draw[trans] (STT.east) --
    node[lbl, below, pos=0.5]{transcription ready} (PLAN.west);

% LLM Planning → Executing Actions  (vertical; label to the left)
\draw[trans] (PLAN.south) --
    node[lbl, left, pos=0.5]{plan generated} (EXEC.north);

% LLM Planning → Error State  (horizontal)
\draw[trans] (PLAN.east) -- node[lbl, above]{LLM error / timeout} (ERR.west);

% Executing Actions ↔ Waiting for Page Load
\draw[trans] (EXEC.east) to[bend left=20]
    node[lbl, above]{await navigation} (WAIT.west);
\draw[trans] (WAIT.west) to[bend left=20]
    node[lbl, below]{page loaded} (EXEC.east);

% Executing Actions → Completed
\draw[trans] (EXEC.south) -- node[lbl, right]{all actions done} (DONE.north);

% Executing Actions → Error State  (CDP error: up from EXEC then right to ERR)
\draw[trans] (EXEC.north east) -- ++(0, 1.0) -- ++(6.2, 0) --
    node[lbl, right]{CDP error} (ERR.south);

% Error State → Receiving Input  (user retries: up then left to RECV.east)
\draw[trans] (ERR.north) -- ++(0, 1.5) -|
    node[lbl, above, pos=0.25]{user retries} (RECV.east);

% Error State → Idle  (abort: right-side bypass, ERR east → far right → up → IDLE east)
\draw[trans] (ERR.east) -- ++(2.0, 0) |-
    node[lbl, above, pos=0.7]{abort} (IDLE.east);

% Completed → Final
\draw[trans] (DONE) -- (F);

% Completed → Idle  (new command: deep-left bypass, well clear of STT box at x≈-9.4)
\draw[trans] (DONE.west) -- ++(-11.0, 0) |-
    node[lbl, right, pos=0.97]{new command} (IDLE.west);

\end{tikzpicture}}
\caption{State Chart Diagram -- AI Agent Command Lifecycle}
\label{fig:statechart}
\end{figure}

\section{Flowchart Diagram}
Figure~\ref{fig:flowchart} presents the end-to-end processing flow for a user command in Prism AI Browser (voice or text).

\begin{figure}[H]
\centering
\scalebox{0.82}{%
\begin{tikzpicture}[
    term/.style={draw, rounded rectangle, minimum width=4.2cm,
                 minimum height=0.85cm, align=center, fill=green!15, font=\small},
    proc/.style={draw, rectangle, minimum width=4.2cm,
                 minimum height=0.85cm, align=center, fill=blue!10, font=\small},
    dec/.style={draw, diamond, aspect=2.4, minimum width=4.2cm,
                minimum height=0.85cm, align=center, fill=yellow!20, font=\small},
    io/.style={draw, trapezium, trapezium left angle=70, trapezium right angle=110,
               minimum width=4.2cm, minimum height=0.85cm,
               align=center, fill=orange!15, font=\small},
    arr/.style={-Stealth, thick}
]

% ---- Nodes (top to bottom, centred at x = 0) ----
% Inter-node gap: 1.8 cm  (was 2.2 cm)
\node[term]  (start)   at (0,   0.0) {Start};
\node[io]    (input)   at (0,  -1.8) {User Input\\(text or voice)};
\node[dec]   (voice)   at (0,  -3.8) {Voice Input?};
\node[proc]  (stt)     at (-5.5,-5.8){Run STT Module\\(audio $\rightarrow$ text)};
\node[proc]  (clean)   at (0,  -5.8) {Sanitise \& Prepare\\Command String};
\node[proc]  (llm)     at (0,  -7.8) {Send to Local LLM\\(generate action plan)};
\node[dec]   (valid)   at (0,  -9.8) {Plan Valid?};
\node[proc]  (retry)   at (5.5, -9.8) {Report Error\\to User};
\node[proc]  (cdpconn) at (0, -11.8) {Connect to CDP\\(ws://127.0.0.1:9222)};
\node[proc]  (exec)    at (0, -13.8) {Execute Next Action\\(navigate / click / eval)};
\node[dec]   (more)    at (0, -15.8) {More Actions?};
\node[proc]  (result)  at (0, -17.8) {Collect Result \&\\Compose Response};
\node[io]    (output)  at (0, -19.8) {Display Response\\in Agent UI};
\node[term]  (end)     at (0, -21.6) {End};

% ---- Arrows ----
\draw[arr] (start)   -- (input);
\draw[arr] (input)   -- (voice);

% Voice branch to STT: horizontal left then vertical down into STT top
\draw[arr] (voice.west) --
    node[above, font=\scriptsize]{Yes}
    (stt |- voice.west) -- (stt.north);

% STT rejoins at clean
\draw[arr] (stt.east) |- (clean.west);

% No-voice path straight to clean
\draw[arr] (voice.south) -- node[right, font=\scriptsize]{No} (clean.north);

\draw[arr] (clean)   -- (llm);
\draw[arr] (llm)     -- (valid);

% Invalid plan → retry (right)
\draw[arr] (valid.east) -- node[above, font=\scriptsize]{No} (retry.west);

% Retry loops back to input via far right
\draw[arr] (retry.east) -- ++(0.6, 0) |-
    node[right, near start, font=\scriptsize]{retry} (input.east);

% Valid plan → connect CDP
\draw[arr] (valid.south) -- node[right, font=\scriptsize]{Yes} (cdpconn.north);

\draw[arr] (cdpconn) -- (exec);
\draw[arr] (exec)    -- (more);

% No more actions → collect result
\draw[arr] (more.south) -- node[right, font=\scriptsize]{No} (result.north);

% More actions → loop back to exec (right side)
\draw[arr] (more.east) -- ++(1.8, 0)
    node[above, font=\scriptsize, pos=0.4]{Yes}
    |- (exec.east);

\draw[arr] (result)  -- (output);
\draw[arr] (output)  -- (end);

\end{tikzpicture}}
\caption{Flowchart -- User Command Processing in Prism AI Browser}
\label{fig:flowchart}
\end{figure}
