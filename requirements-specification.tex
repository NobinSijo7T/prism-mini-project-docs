\chapter{Requirements Specification}

\section{External Interface Requirements}

\subsection{User Interfaces}
\begin{itemize}[leftmargin=*]
    \item \textbf{Prism AI Toolbar Button:} A prism-shaped triangle icon button is embedded in the Chromium toolbar between the Home button and the tab strip. Clicking it opens the AI agent web interface at \texttt{http://127.0.0.1:7788} in a new foreground tab.
    \item \textbf{AI Agent Web UI:} A local web-based chat interface served by the Python backend. Users type natural language commands or activate voice input. The interface displays AI responses, progress indicators for multi-step tasks, and results.
    \item \textbf{Voice Input:} A microphone button within the AI agent interface activates Speech-to-Text (STT) capture. Recognised text is pre-filled into the command input field.
    \item \textbf{Browser Standard UI:} All standard Chromium browser UI elements (address bar, back/forward/reload navigation, bookmarks, extensions toolbar) remain intact.
\end{itemize}

\subsection{Hardware Interfaces}
\begin{itemize}[leftmargin=*]
    \item Microphone: For voice command input via the STT subsystem.
    \item Display: Standard monitor supporting at minimum 1024$\times$768 resolution.
    \item Network Interface: Required for standard web browsing; AI agent operates on the loopback interface only.
\end{itemize}

\subsection{Software Interfaces}
\begin{itemize}[leftmargin=*]
    \item \textbf{Chrome DevTools Protocol (CDP):} The AI agent connects to the Chromium browser via WebSocket on port 9222. CDP provides APIs for target management (\texttt{Target.*}), page navigation (\texttt{Page.*}), DOM manipulation (\texttt{DOM.*}), input simulation (\texttt{Input.*}), and JavaScript execution (\texttt{Runtime.*}).
    \item \textbf{LLM Inference API:} The AI agent backend communicates with a local LLM (e.g., via Ollama or llama.cpp HTTP API) for language understanding and response generation.
    \item \textbf{Web Speech API / STT Engine:} Used for voice-to-text transcription, either via the browser's built-in Web Speech API or a locally hosted STT service.
\end{itemize}

\subsection{Communication Interfaces}
\begin{itemize}[leftmargin=*]
    \item HTTP on \texttt{127.0.0.1:7788}: Browser-to-agent communication for submitting commands.
    \item WebSocket on \texttt{127.0.0.1:9222}: Agent-to-browser CDP communication for automation.
    \item All external internet communication is solely for normal web browsing and uses standard HTTP/HTTPS.
\end{itemize}
